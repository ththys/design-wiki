// 팀원 이메일은 여기(공개 저장소)가 아니라 콘솔에만 적어요.

    function isWikiTeam() {
      return request.auth != null && request.auth.token.email in [
        'me@example.com',
        'teammate1@example.com'
      ];
    }
    match /wiki_pages/{id} {
      allow read, write: if isWikiTeam();
    }
    match /wiki_history/{id} {
      allow read, create: if isWikiTeam();
    }
    match /wiki_comments/{id} {
      allow read: if isWikiTeam();
      allow create: if isWikiTeam() && request.resource.data.by.email == request.auth.token.email;
      allow delete: if isWikiTeam() && resource.data.by.email == request.auth.token.email;
    }
