// 관리자 이메일은 여기(공개 저장소)가 아니라 콘솔에만 적어요. 팀원은 위키의 "팀원 관리"에서 승인해요.

    function isWikiAdmin() {
      return request.auth != null && request.auth.token.email in [
        'me@example.com'
      ];
    }
    function isWikiTeam() {
      return isWikiAdmin() || (request.auth != null && request.auth.token.email_verified == true
        && exists(/databases/$(database)/documents/wiki_members/$(request.auth.token.email)));
    }
    match /wiki_pages/{id} {
      allow read, write: if isWikiTeam();
    }
    match /wiki_history/{id} {
      allow read, create: if isWikiTeam();
    }
    match /wiki_comments/{id} {
      allow read: if isWikiTeam();
      allow create: if isWikiTeam() && request.resource.data.by.email == request.auth.token.email;
      allow delete: if isWikiTeam() && resource.data.by.email == request.auth.token.email;
    }
    match /wiki_members/{email} {
      allow read: if isWikiAdmin() || (request.auth != null && request.auth.token.email == email);
      allow write: if isWikiAdmin();
    }
    match /wiki_requests/{uid} {
      allow read: if isWikiAdmin() || (request.auth != null && request.auth.uid == uid);
      allow create: if request.auth != null && request.auth.uid == uid && request.resource.data.email == request.auth.token.email;
      allow update: if request.auth != null && request.auth.uid == uid && request.resource.data.email == resource.data.email;
      allow delete: if isWikiAdmin();
    }
    // 하루 가입 요청 개수 (3개까지)
    match /wiki_meta/signup {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.resource.data.count is int && request.resource.data.count <= 3
        && (resource == null || resource.data.day != request.resource.data.day
            ? request.resource.data.count == 1
            : request.resource.data.count == resource.data.count + 1);
    }
