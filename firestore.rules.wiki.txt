// 아래 내용을 Firebase 콘솔 > Firestore > 규칙 의 "match /databases/{database}/documents {" 안쪽,
// 기존 monsters / site / portfolio 규칙 옆에 붙여 넣으세요. (기존 규칙은 지우지 마세요)
// 관리자 이메일은 여기(공개 저장소)가 아니라 콘솔에만 적어요. 팀원은 위키의 "팀원 관리"에서 승인해요.

    function isWikiAdmin() {
      return request.auth != null && request.auth.token.email in [
        'me@example.com'
      ];
    }
    function isWikiTeam() {
      return isWikiAdmin() || (request.auth != null
        && exists(/databases/$(database)/documents/wiki_members/$(request.auth.token.email)));
    }
    match /wiki_pages/{id} {
      allow read, write: if isWikiTeam();
    }
    match /wiki_history/{id} {
      allow read, create: if isWikiTeam();
      // 영구 삭제한 페이지의 기록 정리
      allow delete: if isWikiTeam() && !exists(/databases/$(database)/documents/wiki_pages/$(resource.data.pageId));
    }
    match /wiki_comments/{id} {
      allow read: if isWikiTeam();
      allow create: if isWikiTeam() && request.resource.data.by.email == request.auth.token.email;
      // 편집(본인 글은 화면에서만 편집 단추), 할 일 체크·반응은 팀 누구나. 작성자·페이지·시간은 못 바꿈
      allow update: if isWikiTeam() && request.resource.data.by == resource.data.by
        && request.resource.data.pageId == resource.data.pageId && request.resource.data.at == resource.data.at;
      allow delete: if isWikiTeam() && (resource.data.by.email == request.auth.token.email || isWikiAdmin()
        || !exists(/databases/$(database)/documents/wiki_pages/$(resource.data.pageId)));
    }
    // 알림 (@멘션·답글·내 페이지 댓글): 받는 사람만 읽고 지움
    match /wiki_notes/{id} {
      allow read, update, delete: if request.auth != null && resource.data.to == request.auth.token.email.lower();
      allow create: if isWikiTeam() && request.resource.data.from.email == request.auth.token.email;
    }
    // @멘션에 쓰는 팀원 이름 목록
    match /wiki_meta/people {
      allow read, write: if isWikiTeam();
    }
    match /wiki_members/{email} {
      allow read: if isWikiAdmin() || (request.auth != null && request.auth.token.email == email);
      allow write: if isWikiAdmin();
    }
    function wikiSignupOpen() {
      return !exists(/databases/$(database)/documents/wiki_meta/settings)
        || get(/databases/$(database)/documents/wiki_meta/settings).data.get('signupClosed', false) == false;
    }
    match /wiki_meta/settings {
      allow read: if true;
      allow write: if isWikiAdmin();
    }
    match /wiki_requests/{uid} {
      allow read: if isWikiAdmin() || (request.auth != null && request.auth.uid == uid);
      allow create: if wikiSignupOpen() && request.auth != null && request.auth.uid == uid && request.resource.data.email == request.auth.token.email;
      allow update: if request.auth != null && request.auth.uid == uid && request.resource.data.email == resource.data.email;
      allow delete: if isWikiAdmin();
    }
    // 하루 가입 요청 개수 (3개까지)
    match /wiki_meta/signup {
      allow read: if request.auth != null;
      allow write: if wikiSignupOpen() && request.auth != null && request.resource.data.count is int && request.resource.data.count <= 3
        && ((resource == null && request.resource.data.count == 1)
         || (resource != null && resource.data.day != request.resource.data.day && request.resource.data.count == 1)
         || (resource != null && resource.data.day == request.resource.data.day && request.resource.data.count == resource.data.count + 1));
    }
